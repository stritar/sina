// The model builds the dashboard as intent. This schema validates it,
// server-side, and drops rows the viewer isn't cleared to see.
export const reorderDashboard = z
.object({
rows: z.array(reorderRow),
viewer: viewerContext,
})
.strict(); // reject any key not in the schema — no smuggled columns
// Post-parse policy: the need-to-know gate.
function clearancePolicy(data, ctx): Redaction[] {
return data.rows
.filter((row) => outranks(row.classification, data.viewer.clearance))
.map((row) => ({
code: "CLEARANCE_REQUIRED",
message: "export-controlled part — viewer not cleared",
standard: "SINA need-to-know — clearance filter",
severity: "redact", // ← drop the row, keep the dashboard, log it
}));
}
intercept(reorderDashboard, intent) → withheld: 1 row audit: logged mounts: "ReorderDashboard"

The model builds the dashboard. The constitution decides who sees what. An export-controlled row the viewer isn't cleared for never reaches the screen: it's dropped on the server, and an audit line records that it was withheld.
